159template <Direction direction>
160constexpr uint64_t shift(const uint64_t bb) {
161 switch (direction) {
162 case NORTH:
163 return shiftNorth(bb);
164 case SOUTH:
165 return shiftSouth(bb);
166 case EAST:
167 return shiftEast(bb);
168 case WEST:
169 return shiftWest(bb);
170 case NORTH_EAST:
171 return shiftNorthEast(bb);
172 case NORTH_WEST:
173 return shiftNorthWest(bb);
174 case SOUTH_EAST:
175 return shiftSouthEast(bb);
176 case SOUTH_WEST:
177 return shiftSouthWest(bb);
178 case NORTH_NORTH_EAST:
179 return shiftNorthNorthEast(bb);
180 case NORTH_EAST_EAST:
181 return shiftNorthEastEast(bb);
182 case SOUTH_EAST_EAST:
183 return shiftSouthEastEast(bb);
184 case SOUTH_SOUTH_EAST:
185 return shiftSouthSouthEast(bb);
186 case SOUTH_SOUTH_WEST:
187 return shiftSouthSouthWest(bb);
188 case SOUTH_WEST_WEST:
189 return shiftSouthWestWest(bb);
190 default:
191 assert(false);
192 return bb;
193 }
194}
