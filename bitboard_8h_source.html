191template <Direction direction>
192constexpr uint64_t shift(const uint64_t bb) {
193 switch (direction) {
194 case NORTH:
195 return shiftNorth(bb);
196 case SOUTH:
197 return shiftSouth(bb);
198 case EAST:
199 return shiftEast(bb);
200 case WEST:
201 return shiftWest(bb);
202 case NORTH_EAST:
203 return shiftNorthEast(bb);
204 case NORTH_WEST:
205 return shiftNorthWest(bb);
206 case SOUTH_EAST:
207 return shiftSouthEast(bb);
208 case SOUTH_WEST:
209 return shiftSouthWest(bb);
210 case NORTH_NORTH_EAST:
211 return shiftNorthNorthEast(bb);
212 case NORTH_EAST_EAST:
213 return shiftNorthEastEast(bb);
214 case SOUTH_EAST_EAST:
215 return shiftSouthEastEast(bb);
216 case SOUTH_SOUTH_EAST:
217 return shiftSouthSouthEast(bb);
218 case SOUTH_SOUTH_WEST:
219 return shiftSouthSouthWest(bb);
220 case SOUTH_WEST_WEST:
221 return shiftSouthWestWest(bb);
222 default:
223 assert(false);
224 return bb;
225 }
226}
