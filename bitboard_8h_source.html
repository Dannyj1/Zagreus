164template <Direction direction>
165constexpr uint64_t shift(const uint64_t bb) {
166 switch (direction) {
167 case NORTH:
168 return shiftNorth(bb);
169 case SOUTH:
170 return shiftSouth(bb);
171 case EAST:
172 return shiftEast(bb);
173 case WEST:
174 return shiftWest(bb);
175 case NORTH_EAST:
176 return shiftNorthEast(bb);
177 case NORTH_WEST:
178 return shiftNorthWest(bb);
179 case SOUTH_EAST:
180 return shiftSouthEast(bb);
181 case SOUTH_WEST:
182 return shiftSouthWest(bb);
183 case NORTH_NORTH_EAST:
184 return shiftNorthNorthEast(bb);
185 case NORTH_EAST_EAST:
186 return shiftNorthEastEast(bb);
187 case SOUTH_EAST_EAST:
188 return shiftSouthEastEast(bb);
189 case SOUTH_SOUTH_EAST:
190 return shiftSouthSouthEast(bb);
191 case SOUTH_SOUTH_WEST:
192 return shiftSouthSouthWest(bb);
193 case SOUTH_WEST_WEST:
194 return shiftSouthWestWest(bb);
195 default:
196 assert(false);
197 return bb;
198 }
199}
