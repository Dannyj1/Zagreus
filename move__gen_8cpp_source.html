42template <PieceColor color, GenerationType type>
43void generateMoves(const Board& board, MoveList& moves) {
44 assert(moves.size == 0);
45
46 constexpr Piece ownKing = color == WHITE ? WHITE_KING : BLACK_KING;
47 constexpr Piece opponentKing = color == WHITE ? BLACK_KING : WHITE_KING;
48 constexpr PieceColor opponentColor = !color;
49 bool onlyKingMoves = false;
50 const uint64_t ownPieces = board.getColorBitboard<color>();
51 const uint64_t opponentKingBB = board.getPieceBoard<opponentKing>();
52 uint64_t genMask = ~(ownPieces | opponentKingBB);
53
54 if (type == QSEARCH) {
55 const uint64_t opponentPieces = board.getColorBitboard<opponentColor>();
56
57 genMask &= opponentPieces;
58 } else if (type == EVASIONS) {
59 const Square kingSquare = bitboardToSquare(board.getPieceBoard<ownKing>());
60 const uint64_t attackers = board.getSquareAttackersByColor<opponentColor>(kingSquare);
61
62 if (popcnt(attackers) == 1) {
63 const Square attackerSquare = bitboardToSquare(attackers);
64 const PieceType attackerType = getPieceType(board.getPieceOnSquare(attackerSquare));
65 uint64_t evasionsMask = attackers;
66
67 if (isSlidingPiece(attackerType)) {
68 const uint64_t squaresBetween = getSquaresBetween(attackerSquare, kingSquare);
69
70 evasionsMask |= squaresBetween;
71 }
72
73 genMask = evasionsMask;
74 } else {
75 // If the king is in double check, only king moves are legal
76 onlyKingMoves = true;
77 }
78 }
79
80 if (!onlyKingMoves) {
81 generatePawnMoves<color, type>(board, moves, genMask);
82 generateKnightMoves<color, type>(board, moves, genMask);
83 generateBishopMoves<color, type>(board, moves, genMask);
84 generateRookMoves<color, type>(board, moves, genMask);
85 generateQueenMoves<color, type>(board, moves, genMask);
86 }
87
88 if (type == EVASIONS) {
89 // We generate all king moves when in check, so we don't use genMask
90 generateKingMoves<color, type>(board, moves, ~(ownPieces | opponentKingBB));
91 } else {
92 generateKingMoves<color, type>(board, moves, genMask);
93 }
94
95 assert(type == QSEARCH || genMask != 0);
96 assert((genMask & ownPieces) == 0);
97 assert((genMask & opponentKingBB) == 0);
98}

108template <PieceColor color, GenerationType type>
109void generatePawnMoves(const Board& board, MoveList& moves, const uint64_t genMask) {
110 constexpr Piece pawn = color == WHITE ? WHITE_PAWN : BLACK_PAWN;
111 constexpr PieceColor opponentColor = !color;
112
113 const uint64_t pawnBB = board.getPieceBoard<pawn>();
114 const uint64_t emptyBB = board.getEmptyBitboard();
115 const uint64_t opponentPieces = board.getColorBitboard<opponentColor>();
116 uint64_t pawnSinglePushes;
117 uint64_t pawnDoublePushes;
118 uint64_t pawnWestAttacks;
119 uint64_t pawnEastAttacks;
120
121 if constexpr (color == WHITE) {
122 pawnSinglePushes = whitePawnSinglePush(pawnBB, emptyBB);
123 pawnDoublePushes = whitePawnDoublePush(pawnBB, emptyBB);
124 pawnWestAttacks = whitePawnWestAttacks(pawnBB);
125 pawnEastAttacks = whitePawnEastAttacks(pawnBB);
126 } else {
127 pawnSinglePushes = blackPawnSinglePush(pawnBB, emptyBB);
128 pawnDoublePushes = blackPawnDoublePush(pawnBB, emptyBB);
129 pawnWestAttacks = blackPawnWestAttacks(pawnBB);
130 pawnEastAttacks = blackPawnEastAttacks(pawnBB);
131 }
132
133 uint64_t enPassantMask = squareToBitboard(board.getEnPassantSquare());
134
135 if constexpr (color == WHITE) {
136 enPassantMask &= RANK_6;
137 } else {
138 enPassantMask &= RANK_3;
139 }
140
141 pawnSinglePushes &= genMask;
142 pawnDoublePushes &= genMask;
143 pawnWestAttacks &= (opponentPieces | enPassantMask) & genMask;
144 pawnEastAttacks &= (opponentPieces | enPassantMask) & genMask;
145
146 constexpr Direction fromPushDirection = color == WHITE ? NORTH : SOUTH;
147 constexpr Direction fromSqWestAttackDirection = color == WHITE ? NORTH_WEST : SOUTH_WEST;
148 constexpr Direction fromSqEastAttackDirection = color == WHITE ? NORTH_EAST : SOUTH_EAST;
149 constexpr uint64_t promotionRank = color == WHITE ? RANK_8 : RANK_1;
150
151 while (pawnSinglePushes) {
152 const uint8_t squareTo = popLsb(pawnSinglePushes);
153 const uint64_t squareToBB = squareToBitboard(squareTo);
154 const uint8_t squareFrom = squareTo - fromPushDirection;
155
156 if (squareToBB & promotionRank) {
157 for (const PromotionPiece promotionPiece :
158 {QUEEN_PROMOTION, ROOK_PROMOTION, BISHOP_PROMOTION, KNIGHT_PROMOTION}) {
159 const Move move = encodeMove(squareFrom, squareTo, promotionPiece);
160 moves.moves[moves.size] = move;
161 moves.size++;
162 }
163 } else {
164 const Move move = encodeMove(squareFrom, squareTo);
165
166 moves.moves[moves.size] = move;
167 moves.size++;
168 }
169 }
170
171 while (pawnDoublePushes) {
172 const uint8_t squareTo = popLsb(pawnDoublePushes);
173 const uint8_t squareFrom = squareTo - fromPushDirection - fromPushDirection;
174 const Move move = encodeMove(squareFrom, squareTo);
175
176 moves.moves[moves.size] = move;
177 moves.size++;
178 }
179
180 while (pawnWestAttacks) {
181 const uint8_t squareTo = popLsb(pawnWestAttacks);
182 const uint64_t squareToBB = squareToBitboard(squareTo);
183 const uint8_t squareFrom = squareTo - fromSqWestAttackDirection;
184
185 if (squareTo == board.getEnPassantSquare() && squareToBB & enPassantMask) {
186 const Move move = encodeMove(squareFrom, squareTo, EN_PASSANT);
187 moves.moves[moves.size] = move;
188 moves.size++;
189 } else {
190 if (squareToBB & promotionRank) {
191 for (const PromotionPiece promotionPiece :
192 {QUEEN_PROMOTION, ROOK_PROMOTION, BISHOP_PROMOTION, KNIGHT_PROMOTION}) {
193 const Move move = encodeMove(squareFrom, squareTo, promotionPiece);
194 moves.moves[moves.size] = move;
195 moves.size++;
196 }
197 } else {
198 const Move move = encodeMove(squareFrom, squareTo);
199 moves.moves[moves.size] = move;
200 moves.size++;
201 }
202 }
203 }
204
205 while (pawnEastAttacks) {
206 const uint8_t squareTo = popLsb(pawnEastAttacks);
207 const uint64_t squareToBB = squareToBitboard(squareTo);
208 const uint8_t squareFrom = squareTo - fromSqEastAttackDirection;
209
210 if (squareTo == board.getEnPassantSquare() && squareToBB & enPassantMask) {
211 const Move move = encodeMove(squareFrom, squareTo, EN_PASSANT);
212 moves.moves[moves.size] = move;
213 moves.size++;
214 } else {
215 if (squareToBB & promotionRank) {
216 for (const PromotionPiece promotionPiece :
217 {QUEEN_PROMOTION, ROOK_PROMOTION, BISHOP_PROMOTION, KNIGHT_PROMOTION}) {
218 const Move move = encodeMove(squareFrom, squareTo, promotionPiece);
219 moves.moves[moves.size] = move;
220 moves.size++;
221 }
222 } else {
223 const Move move = encodeMove(squareFrom, squareTo);
224 moves.moves[moves.size] = move;
225 moves.size++;
226 }
227 }
228 }
229}

239template <PieceColor color, GenerationType type>
240void generateKnightMoves(const Board& board, MoveList& moves, const uint64_t genMask) {
241 constexpr Piece knight = color == WHITE ? WHITE_KNIGHT : BLACK_KNIGHT;
242 uint64_t knightBB = board.getPieceBoard<knight>();
243
244 while (knightBB) {
245 const uint8_t fromSquare = popLsb(knightBB);
246 const uint64_t attacks = getKnightAttacks(fromSquare);
247 uint64_t genBB = attacks & genMask;
248
249 while (genBB) {
250 const uint8_t toSquare = popLsb(genBB);
251 const Move move = encodeMove(fromSquare, toSquare);
252
253 moves.moves[moves.size] = move;
254 moves.size++;
255 }
256 }

351template <PieceColor color, GenerationType type>
352void generateKingMoves(const Board& board, MoveList& moves, const uint64_t genMask) {
353 constexpr Piece king = color == WHITE ? WHITE_KING : BLACK_KING;
354 uint64_t kingBB = board.getPieceBoard<king>();
355 const uint8_t fromSquare = popLsb(kingBB);
356 uint64_t genBB = getKingAttacks(fromSquare) & genMask;
357
358 while (genBB) {
359 const uint8_t toSquare = popLsb(genBB);
360 const Move move = encodeMove(fromSquare, toSquare);
361
362 moves.moves[moves.size] = move;
363 moves.size++;
364 }
365
366 const uint8_t castlingRights = board.getCastlingRights();
367
368 if constexpr (color == WHITE) {
369 if (castlingRights & WHITE_KINGSIDE && board.canCastle<WHITE_KINGSIDE>()) {
370 const Move move = encodeMove(E1, G1, CASTLING);
371
372 moves.moves[moves.size] = move;
373 moves.size++;
374 }
375
376 if (castlingRights & WHITE_QUEENSIDE && board.canCastle<WHITE_QUEENSIDE>()) {
377 const Move move = encodeMove(E1, C1, CASTLING);
378
379 moves.moves[moves.size] = move;
380 moves.size++;
381 }
382 } else if constexpr (color == BLACK) {
383 if (castlingRights & BLACK_KINGSIDE && board.canCastle<BLACK_KINGSIDE>()) {
384 const Move move = encodeMove(E8, G8, CASTLING);
385
386 moves.moves[moves.size] = move;
387 moves.size++;
388 }
389
390 if (castlingRights & BLACK_QUEENSIDE && board.canCastle<BLACK_QUEENSIDE>()) {
391 const Move move = encodeMove(E8, C8, CASTLING);
392
393 moves.moves[moves.size] = move;
394 moves.size++;
395 }
