24/*
25 * This code provides the reference implementation of the PCG family of
26 * random number generators. The code is complex because it implements
27 *
28 * - several members of the PCG family, specifically members corresponding
29 * to the output functions:
30 * - XSH RR (good for 64-bit state, 32-bit output)
31 * - XSH RS (good for 64-bit state, 32-bit output)
32 * - XSL RR (good for 128-bit state, 64-bit output)
33 * - RXS M XS (statistically most powerful generator)
34 * - XSL RR RR (good for 128-bit state, 128-bit output)
35 * - and RXS, RXS M, XSH, XSL (mostly for testing)
36 * - at potentially *arbitrary* bit sizes
37 * - with four different techniques for random streams (MCG, one-stream
38 * LCG, settable-stream LCG, unique-stream LCG)
39 * - and the extended generation schemes allowing arbitrary periods
40 * - with all features of C++11 random number generation (and more),
41 * some of which are somewhat painful, including
42 * - initializing with a SeedSequence which writes 32-bit values
43 * to memory, even though the state of the generator may not
44 * use 32-bit values (it might use smaller or larger integers)
45 * - I/O for RNGs and a prescribed format, which needs to handle
46 * the issue that 8-bit and 128-bit integers don't have working
47 * I/O routines (e.g., normally 8-bit = char, not integer)
48 * - equality and inequality for RNGs
49 * - and a number of convenience typedefs to mask all the complexity
50 *
51 * The code employes a fairly heavy level of abstraction, and has to deal
52 * with various C++ minutia. If you're looking to learn about how the PCG
53 * scheme works, you're probably best of starting with one of the other
54 * codebases (see www.pcg-random.org). But if you're curious about the
55 * constants for the various output functions used in those other, simpler,
56 * codebases, this code shows how they are calculated.
57 *
58 * On the positive side, at least there are convenience typedefs so that you
59 * can say
60 *
61 * pcg32 myRNG;
62 *
63 * rather than:
64 *
65 * pcg_detail::engine<
66 * uint32_t, // Output Type
67 * uint64_t, // State Type
68 * pcg_detail::xsh_rr_mixin<uint32_t, uint64_t>, true, // Output Func
69 * pcg_detail::specific_stream<uint64_t>, // Stream Kind
70 * pcg_detail::default_multiplier<uint64_t> // LCG Mult
71 * > myRNG;
72 *
73 */

154/*
155 * Each PCG generator is available in four variants, based on how it applies
156 * the additive constant for its underlying LCG; the variations are:
157 *
158 * single stream - all instances use the same fixed constant, thus
159 * the RNG always somewhere in same sequence
160 * mcg - adds zero, resulting in a single stream and reduced
161 * period
162 * specific stream - the constant can be changed at any time, selecting
163 * a different random sequence
164 * unique stream - the constant is based on the memory addresss of the
165 * object, thus every RNG has its own unique sequence
166 *
167 * This variation is provided though mixin classes which define a function
168 * value called increment() that returns the nesessary additive constant.
169 */
170

283/*
284 * This is where it all comes together. This function joins together three
285 * mixin classes which define
286 * - the LCG additive constant (the stream)
287 * - the LCG multiplier
288 * - the output function
289 * in addition, we specify the type of the LCG state, and the result type,
290 * and whether to use the pre-advance version of the state for the output
291 * (increasing instruction-level parallelism) or the post-advance version
292 * (reducing register pressure).
293 *
294 * Given the high level of parameterization, the code has to use some
295 * template-metaprogramming tricks to handle some of the suble variations
296 * involved.
297 */

453template <typename CharT, typename Traits, typename xtype, typename itype, typename output_mixin, bool output_previous,
454 typename stream_mixin, typename multiplier_mixin>
455std::basic_istream<CharT, Traits>& operator>>(
456 std::basic_istream<CharT, Traits>& in,
457 engine<xtype, itype, output_mixin, output_previous, stream_mixin, multiplier_mixin>& rng) {
458 auto orig_flags = in.flags(std::ios_base::dec | std::ios_base::skipws);
459
460 itype multiplier, increment, state;
461 in >> multiplier >> increment >> state;
462
463 if (!in.fail()) {
464 bool good = true;
465 if (multiplier != rng.multiplier()) {
466 good = false;
467 } else if (rng.can_specify_stream) {
468 rng.set_stream(increment >> 1);
469 } else if (increment != rng.increment()) {
470 good = false;
471 }
472 if (good) {
473 rng.state_ = state;
474 } else {
475 in.clear(std::ios::failbit);
476 }
477 }
478
479 in.flags(orig_flags);
480 return in;
481}

483template <typename xtype, typename itype, typename output_mixin, bool output_previous, typename stream_mixin,
484 typename multiplier_mixin>
485itype engine<xtype, itype, output_mixin, output_previous, stream_mixin, multiplier_mixin>::advance(itype state,
486 itype delta,
487 itype cur_mult,
488 itype cur_plus) {
489 // The method used here is based on Brown, "Random Number Generation
490 // with Arbitrary Stride,", Transactions of the American Nuclear
491 // Society (Nov. 1994). The algorithm is very similar to fast
492 // exponentiation.
493 //
494 // Even though delta is an unsigned integer, we can pass a
495 // signed integer to go backwards, it just goes "the long way round".
496
497 constexpr itype ZERO = 0u; // itype may be a non-trivial types, so
498 constexpr itype ONE = 1u; // we define some ugly constants.
499 itype acc_mult = 1;
500 itype acc_plus = 0;
501 while (delta > ZERO) {
502 if (delta & ONE) {
503 acc_mult *= cur_mult;
504 acc_plus = acc_plus * cur_mult + cur_plus;
505 }
506 cur_plus = (cur_mult + ONE) * cur_plus;
507 cur_mult *= cur_mult;
508 delta >>= 1;
509 }
510 return acc_mult * state + acc_plus;
511}

513template <typename xtype, typename itype, typename output_mixin, bool output_previous, typename stream_mixin,
514 typename multiplier_mixin>
515itype engine<xtype, itype, output_mixin, output_previous, stream_mixin, multiplier_mixin>::distance(
516 itype cur_state, itype newstate, itype cur_mult, itype cur_plus, itype mask) {
517 constexpr itype ONE = 1u; // itype could be weird, so use constant
518 itype the_bit = stream_mixin::is_mcg ? itype(4u) : itype(1u);
519 itype distance = 0u;
520 while ((cur_state & mask) != (newstate & mask)) {
521 if ((cur_state & the_bit) != (newstate & the_bit)) {
522 cur_state = cur_state * cur_mult + cur_plus;
523 distance |= the_bit;
524 }
525 assert((cur_state & the_bit) == (newstate & the_bit));
526 the_bit <<= 1;
527 cur_plus = (cur_mult + ONE) * cur_plus;
528 cur_mult *= cur_mult;
529 }
530 return stream_mixin::is_mcg ? distance >> 2 : distance;
531}

575/*
576 * OUTPUT FUNCTIONS.
577 *
578 * These are the core of the PCG generation scheme. They specify how to
579 * turn the base LCG's internal state into the output value of the final
580 * generator.
581 *
582 * They're implemented as mixin classes.
583 *
584 * All of the classes have code that is written to allow it to be applied
585 * at *arbitrary* bit sizes, although in practice they'll only be used at
586 * standard sizes supported by C++.
587 */
588
589/*
590 * XSH RS -- high xorshift, followed by a random shift
591 *
592 * Fast. A good performer.
593 */
594
595template <typename xtype, typename itype>
596struct xsh_rs_mixin {
597 static xtype output(itype internal) {
598 constexpr bitcount_t bits = bitcount_t(sizeof(itype) * 8);
599 constexpr bitcount_t xtypebits = bitcount_t(sizeof(xtype) * 8);
600 constexpr bitcount_t sparebits = bits - xtypebits;
601 constexpr bitcount_t opbits = sparebits - 5 >= 64 ? 5
602 : sparebits - 4 >= 32 ? 4
603 : sparebits - 3 >= 16 ? 3
604 : sparebits - 2 >= 4 ? 2
605 : sparebits - 1 >= 1 ? 1
606 : 0;
607 constexpr bitcount_t mask = (1 << opbits) - 1;
608 constexpr bitcount_t maxrandshift = mask;
609 constexpr bitcount_t topspare = opbits;
610 constexpr bitcount_t bottomspare = sparebits - topspare;
611 constexpr bitcount_t xshift = topspare + (xtypebits + maxrandshift) / 2;
612 bitcount_t rshift = opbits ? bitcount_t(internal >> (bits - opbits)) & mask : 0;
613 internal ^= internal >> xshift;
614 xtype result = xtype(internal >> (bottomspare - maxrandshift + rshift));
615 return result;
616 }
617};

620 * XSH RR -- high xorshift, followed by a random rotate
621 *
622 * Fast. A good performer. Slightly better statistically than XSH RS.
623 */
624
625template <typename xtype, typename itype>
626struct xsh_rr_mixin {
627 static xtype output(itype internal) {
628 constexpr bitcount_t bits = bitcount_t(sizeof(itype) * 8);

923template <typename baseclass>
924struct inside_out : private baseclass {
925 inside_out() = delete;
926
927 typedef typename baseclass::result_type result_type;
928 typedef typename baseclass::state_type state_type;
929 static_assert(sizeof(result_type) == sizeof(state_type), "Require a RNG whose output function is a permutation");
930
931 static bool external_step(result_type& randval, size_t i) {
932 state_type state = baseclass::unoutput(randval);
933 state = state * baseclass::multiplier() + baseclass::increment() + state_type(i * 2);
934 result_type result = baseclass::output(state);
935 randval = result;
936 state_type zero = baseclass::is_mcg ? state & state_type(3U) : state_type(0U);
937 return result == zero;
938 }
939
940 static bool external_advance(result_type& randval, size_t i, result_type delta, bool forwards = true) {
941 state_type state = baseclass::unoutput(randval);
942 state_type mult = baseclass::multiplier();
943 state_type inc = baseclass::increment() + state_type(i * 2);
944 state_type zero = baseclass::is_mcg ? state & state_type(3U) : state_type(0U);
945 state_type dist_to_zero = baseclass::distance(state, zero, mult, inc);
946 bool crosses_zero = forwards ? dist_to_zero <= delta : (-dist_to_zero) <= delta;
947 if (!forwards) delta = -delta;
948 state = baseclass::advance(state, delta, mult, inc);
949 randval = baseclass::output(state);
950 return crosses_zero;
951 }
952};

1177 typedef typename extvalclass::state_type ext_state_t;

1179 constexpr bitcount_t extbits = sizeof(ext_state_t) * 8;
1180 static_assert(basebits <= extbits || advance_pow2 > 0, "Current implementation might overflow its carry");
1181

1184 base_state_t total_delta = carry + delta;

1186 if (basebits > extbits) {
1187 carry = total_delta >> extbits;

1189 carry = 0;

1192 }

1198 "Efficient advance is too hard for non-kdd extension. "
1199 "For a weak advance, cast to base class");
1200 state_type zero = baseclass::is_mcg ? this->state_ & state_type(3U) : state_type(0U);
1201 if (may_tick) {
1202 state_type ticks = distance >> (advance_pow2 * may_tick);
1203 // ^-- stupidity to appease GCC
1204 // warnings
1205 state_type adv_mask = baseclass::is_mcg ? tick_mask << 2 : tick_mask;
1206 state_type next_advance_distance = this->distance(zero, adv_mask);
1207 if (!forwards) next_advance_distance = (-next_advance_distance) & tick_mask;
1208 if (next_advance_distance < (distance & tick_mask)) {
1209 ++ticks;
1210 }
1211 if (ticks) advance_table(ticks, forwards);
1212 }
1213 if (forwards) {
1214 if (may_tock && this->distance(zero) <= distance) advance_table();
1215 baseclass::advance(distance);
1216 } else {
1217 if (may_tock && -(this->distance(zero)) <= distance) advance_table(state_type(1U), false);
1218 baseclass::advance(-distance);
1219 }
1220}
