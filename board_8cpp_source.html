38void initZobristConstants() {
39 pcg64_oneseq_once_insecure rng;
40
41 rng.seed(0x661778f67199663dULL);
42
43 for (uint64_t& zobrist : zobristConstants) {
44 zobrist = rng();
45
46 while (zobrist == 0) {
47 zobrist = rng();
48 }
49 }
50}

66template <PieceColor movedColor>
67bool Board::isPositionLegal() const {
68 constexpr PieceColor opponentColor = !movedColor;
69 constexpr Piece king = movedColor == WHITE ? WHITE_KING : BLACK_KING;
70 const uint64_t kingBB = getPieceBoard<king>();
71 const Square kingSquare = bitboardToSquare(kingBB);
72 const Move lastMove = getLastMove();
73 const MoveType lastMoveType = getMoveType(lastMove);
74
75 if (lastMoveType == CASTLING) {
76 const Square fromSquare = getFromSquare(lastMove);
77 const Square toSquare = getToSquare(lastMove);
78 const uint64_t fromSquareAttacks = getSquareAttackersByColor<opponentColor>(fromSquare);
79
80 if (fromSquareAttacks) {
81 // King was in check before castling
82 return false;
83 }
84
85 uint64_t castlingPath = 0;
86
87 if (toSquare == G1) {
88 castlingPath = WHITE_KINGSIDE_CASTLE_PATH;
89 } else if (toSquare == C1) {
90 castlingPath = WHITE_QUEENSIDE_CASTLE_PATH;
91 } else if (toSquare == G8) {
92 castlingPath = BLACK_KINGSIDE_CASTLE_PATH;
93 } else if (toSquare == C8) {
94 castlingPath = BLACK_QUEENSIDE_CASTLE_PATH;
95 }
96
97 while (castlingPath) {
98 const Square square = static_cast<Square>(popLsb(castlingPath));
99 const uint64_t attackers = getSquareAttackersByColor<opponentColor>(square);
100
101 if (attackers) {
102 return false;
103 }
104 }
105 }
106
107 return !getSquareAttackersByColor<opponentColor>(kingSquare);
108}

119template <CastlingRights side>
120bool Board::canCastle() const {
121 assert(side != WHITE_CASTLING && side != BLACK_CASTLING);
122
123 if (!(castlingRights & side)) {
124 return false;
125 }
126
127 uint64_t castlingPath = 0;
128
129 if constexpr (side == WHITE_KINGSIDE) {
130 castlingPath = WHITE_KINGSIDE_CASTLE_UNOCCUPIED;
131 } else if constexpr (side == WHITE_QUEENSIDE) {
132 castlingPath = WHITE_QUEENSIDE_CASTLE_UNOCCUPIED;
133 } else if constexpr (side == BLACK_KINGSIDE) {
134 castlingPath = BLACK_KINGSIDE_CASTLE_UNOCCUPIED;
135 } else if constexpr (side == BLACK_QUEENSIDE) {
136 castlingPath = BLACK_QUEENSIDE_CASTLE_UNOCCUPIED;
137 }
138
139 if (occupied & castlingPath) {
140 return false;
141 }
142
143 return true;
144}

156uint64_t Board::getSquareAttackers(const Square square) const {
157 assert(square < SQUARES);
158 const uint64_t knights = getPieceBoard<WHITE_KNIGHT>() | getPieceBoard<BLACK_KNIGHT>();
159 const uint64_t kings = getPieceBoard<WHITE_KING>() | getPieceBoard<BLACK_KING>();
160 uint64_t bishopsQueens = getPieceBoard<WHITE_QUEEN>() | getPieceBoard<BLACK_QUEEN>();
161 uint64_t rooksQueens = getPieceBoard<WHITE_QUEEN>() | getPieceBoard<BLACK_QUEEN>();
162 rooksQueens |= getPieceBoard<WHITE_ROOK>() | getPieceBoard<BLACK_ROOK>();
163 bishopsQueens |= getPieceBoard<WHITE_BISHOP>() | getPieceBoard<BLACK_BISHOP>();
164
165 return (getPawnAttacks<WHITE>(square) & getPieceBoard<BLACK_PAWN>()) |
166 (getPawnAttacks<BLACK>(square) & getPieceBoard<WHITE_PAWN>()) | (getKnightAttacks(square) & knights) |
167 (getKingAttacks(square) & kings) | (getBishopAttacks(square, occupied) & bishopsQueens) |
168 (getRookAttacks(square, occupied) & rooksQueens);
169}

177template <PieceColor color>
178uint64_t Board::getSquareAttackersByColor(const Square square) const {
179 assert(square < SQUARES);
180 constexpr PieceColor opponentColor = !color;
181
182 const uint64_t knights = getPieceBoard<color == WHITE ? WHITE_KNIGHT : BLACK_KNIGHT>();
183 const uint64_t kings = getPieceBoard<color == WHITE ? WHITE_KING : BLACK_KING>();
184 uint64_t bishopsQueens = getPieceBoard<color == WHITE ? WHITE_QUEEN : BLACK_QUEEN>();
185 uint64_t rooksQueens = getPieceBoard<color == WHITE ? WHITE_QUEEN : BLACK_QUEEN>();
186 rooksQueens |= getPieceBoard<color == WHITE ? WHITE_ROOK : BLACK_ROOK>();
187 bishopsQueens |= getPieceBoard<color == WHITE ? WHITE_BISHOP : BLACK_BISHOP>();
188
189 return (getPawnAttacks<opponentColor>(square) & getPieceBoard<color == WHITE ? WHITE_PAWN : BLACK_PAWN>()) |
190 (getKnightAttacks(square) & knights) | (getKingAttacks(square) & kings) |
191 (getBishopAttacks(square, occupied) & bishopsQueens) | (getRookAttacks(square, occupied) & rooksQueens);
192}

238bool Board::isDraw() const {
239 const uint64_t occupied = getOccupiedBitboard();
240
241 if (halfMoveClock >= 100) {
242 return true;
243 }
244
245 const uint64_t zobristHash = getZobristHash();
246
247 // 3-fold repetition
248 int repetitions = 0;
249
250 for (int i = ply - 2; i >= (ply - halfMoveClock); i -= 2) {
251 if (history[i].zobristHash == zobristHash) {
252 repetitions++;
253 }
254
255 // The current board state happens twice in history, making this appearance the third time -> 3-fold repetition
256 if (repetitions >= 2) {
257 return true;
258 }
259 }
260
261 const uint64_t sufficientMaterial = getPieceBoard<WHITE_QUEEN>() | getPieceBoard<BLACK_QUEEN>() |
262 getPieceBoard<WHITE_ROOK>() | getPieceBoard<BLACK_ROOK>() |
263 getPieceBoard<WHITE_PAWN>() | getPieceBoard<BLACK_PAWN>();
264
265 if (sufficientMaterial) {
266 // Can never be a draw with queens, rooks or pawns
267 return false;
268 }
269
270 // Only kings left
271 if (popcnt(occupied) == 2) {
272 return true;
273 }
274
275 // Check for KBvK, KNvK
276 if (popcnt(occupied) == 3) {
277 if (getPieceBoard<WHITE_BISHOP>() || getPieceBoard<BLACK_BISHOP>() || getPieceBoard<WHITE_KNIGHT>() ||
278 getPieceBoard<BLACK_KNIGHT>()) {
279 return true;
280 }
281 }
282
283 // Check for KBvKB where bishops are on the same color, and KNvKN
284 if (popcnt(occupied) == 4) {
285 if (getPieceBoard<WHITE_BISHOP>() && getPieceBoard<BLACK_BISHOP>()) {
286 if (((getPieceBoard<WHITE_BISHOP>() & DARK_SQUARES) && (getPieceBoard<BLACK_BISHOP>() & DARK_SQUARES)) ||
287 ((getPieceBoard<WHITE_BISHOP>() & LIGHT_SQUARES) && (getPieceBoard<BLACK_BISHOP>() & LIGHT_SQUARES))) {
288 return true;
289 }
290 }
291 if (getPieceBoard<WHITE_KNIGHT>() && getPieceBoard<BLACK_KNIGHT>()) {
292 return true;
293 }
294 }
295
296 return false;
297}

308template <PieceColor color>
309Square Board::getSmallestAttacker(Square square) const {
310 const uint64_t attackers = getSquareAttackersByColor<color>(square);
311
312 if (!attackers) {
313 return NONE;
314 }
315
316 constexpr Piece attackerOrder[6] = {
317 color == WHITE ? WHITE_PAWN : BLACK_PAWN, color == WHITE ? WHITE_KNIGHT : BLACK_KNIGHT,
318 color == WHITE ? WHITE_BISHOP : BLACK_BISHOP, color == WHITE ? WHITE_ROOK : BLACK_ROOK,
319 color == WHITE ? WHITE_QUEEN : BLACK_QUEEN, color == WHITE ? WHITE_KING : BLACK_KING};
320
321 for (const Piece piece : attackerOrder) {
322 if (const uint64_t candidate = attackers & bitboards[piece]) {
323 return static_cast<Square>(bitscanForward(candidate));
324 }
325 }
326
327 return NONE;
328}

330int estimateMoveValue(const Board& board, const Move move) {
331 const MoveType moveType = getMoveType(move);
332 Piece capturedPiece = board.getPieceOnSquare(getToSquare(move));
333
334 if (moveType == EN_PASSANT) {
335 capturedPiece = board.getSideToMove() == WHITE ? BLACK_PAWN : WHITE_PAWN;
336 }
337
338 int value = getPieceValue(capturedPiece);
339
340 if (moveType == PROMOTION) {
341 value += getPieceValue(getPieceFromPromotionPiece(getPromotionPiece(move), board.getSideToMove())) -
342 getPieceValue(WHITE_PAWN);
343 } else if (moveType == CASTLING) {
344 value = 0;
345 }
346
347 return value;
348}

358bool Board::see(const Move move, int threshold) {
359 const Square fromSquare = getFromSquare(move);
360 const Square toSquare = getToSquare(move);
361 const MoveType moveType = getMoveType(move);
362 const Piece nextVictim = moveType == PROMOTION ? getPieceFromPromotionPiece(getPromotionPiece(move), sideToMove)
363 : getPieceOnSquare(fromSquare);
364 PieceType nextVictimType = getPieceType(nextVictim);
365 const int moveValue = estimateMoveValue(*this, move);
366 int balance = moveValue - threshold;
367
368 if (balance < 0) {
369 return false;
370 }
371
372 balance -= getPieceValue(nextVictim);
373
374 if (balance >= 0) {
375 return true;
376 }
377
378 const uint64_t bishops = getPieceBoard<WHITE_BISHOP>() | getPieceBoard<BLACK_BISHOP>() |
379 getPieceBoard<WHITE_QUEEN>() | getPieceBoard<BLACK_QUEEN>();
380 const uint64_t rooks = getPieceBoard<WHITE_ROOK>() | getPieceBoard<BLACK_ROOK>() | getPieceBoard<WHITE_QUEEN>() |
381 getPieceBoard<BLACK_QUEEN>();
382 const uint64_t oldOccupied = getOccupiedBitboard();
383 const PieceColor oldSideToMove = sideToMove;
384 occupied = (occupied ^ squareToBitboard(fromSquare)) | squareToBitboard(toSquare);
385
386 if (moveType == EN_PASSANT) {
387 const Square capturedPawnSquare =
388 sideToMove == WHITE ? static_cast<Square>(toSquare + SOUTH) : static_cast<Square>(toSquare + NORTH);
389 occupied ^= squareToBitboard(capturedPawnSquare);
390 }
391
392 uint64_t attackers = getSquareAttackers(toSquare) & occupied;
393 sideToMove = !sideToMove;
394
395 while (true) {
396 const uint64_t ownAttackers = attackers & getColorBitboard(sideToMove);
397
398 if (!ownAttackers) {
399 break;
400 }
401
402 for (nextVictimType = PAWN; nextVictimType <= QUEEN;
403 nextVictimType = static_cast<PieceType>(nextVictimType + 1)) {
404 if (ownAttackers & getPieceBoard(getPieceFromType(nextVictimType, sideToMove))) {
405 break;
406 }
407 }
408
409 occupied ^= squareToBitboard(
410 bitscanForward(ownAttackers & getPieceBoard(getPieceFromType(nextVictimType, sideToMove))));
411
412 if (nextVictimType == PAWN || nextVictimType == BISHOP || nextVictimType == QUEEN) {
413 attackers |= getBishopAttacks(toSquare, occupied) & bishops;
414 }
415
416 if (nextVictimType == ROOK || nextVictimType == QUEEN) {
417 attackers |= getRookAttacks(toSquare, occupied) & rooks;
418 }
419
420 attackers &= occupied;
421 balance = -balance - 1 - getPieceValue(getPieceFromType(nextVictimType, sideToMove));
422 sideToMove = !sideToMove;
423
424 if (balance >= 0) {
425 if (nextVictimType == KING && (attackers & getColorBitboard(sideToMove))) {
426 sideToMove = !sideToMove;
427 }
428
429 break;
430 }
431 }
432
433 const bool result = sideToMove != oldSideToMove;
434 occupied = oldOccupied;
435 sideToMove = oldSideToMove;
436 return result;
437}

450void Board::makeMove(const Move move) {
451 const uint8_t fromSquare = getFromSquare(move);
452 const uint8_t toSquare = getToSquare(move);
453 const MoveType moveType = getMoveType(move);
454 const Piece movedPiece = getPieceOnSquare(fromSquare);
455 const PieceType movedPieceType = getPieceType(movedPiece);
456 Piece capturedPiece = getPieceOnSquare(toSquare);
457
458 if (moveType == EN_PASSANT) {
459 capturedPiece = sideToMove == WHITE ? BLACK_PAWN : WHITE_PAWN;
460 }
461
462 history[ply].move = move;
463 history[ply].previousMove = previousMove;
464 history[ply].capturedPiece = capturedPiece;
465 history[ply].enPassantSquare = enPassantSquare;
466 history[ply].castlingRights = castlingRights;
467 history[ply].zobristHash = zobristHash;
468 history[ply].halfMoveClock = halfMoveClock;
469
470 if (enPassantSquare != NO_EN_PASSANT) {
471 zobristHash ^= getZobristConstant(ZOBRIST_EN_PASSANT_START_INDEX + (enPassantSquare % 8));
472 }
473
474 enPassantSquare = NO_EN_PASSANT;
475
476 halfMoveClock += 1;
477
478 if (capturedPiece != EMPTY && moveType != EN_PASSANT) {
479 removePiece(capturedPiece, toSquare);
480 halfMoveClock = 0;
481
482 if (capturedPiece == WHITE_ROOK) {
483 if (toSquare == A1) {
484 if (castlingRights & WHITE_QUEENSIDE) {
485 castlingRights &= ~WHITE_QUEENSIDE;
486 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_QUEENSIDE_INDEX);
487 }
488 } else if (toSquare == H1) {
489 if (castlingRights & WHITE_KINGSIDE) {
490 castlingRights &= ~WHITE_KINGSIDE;
491 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_KINGSIDE_INDEX);
492 }
493 }
494 } else if (capturedPiece == BLACK_ROOK) {
495 if (toSquare == A8) {
496 if (castlingRights & BLACK_QUEENSIDE) {
497 castlingRights &= ~BLACK_QUEENSIDE;
498 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_QUEENSIDE_INDEX);
499 }
500 } else if (toSquare == H8) {
501 if (castlingRights & BLACK_KINGSIDE) {
502 castlingRights &= ~BLACK_KINGSIDE;
503 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_KINGSIDE_INDEX);
504 }
505 }
506 }
507 }
508
509 removePiece(movedPiece, fromSquare);
510
511 if (moveType == PROMOTION) {
512 const PieceColor color = getPieceColor(movedPiece);
513 const PromotionPiece promotionPieceType = getPromotionPiece(move);
514 const Piece promotionPiece = getPieceFromPromotionPiece(promotionPieceType, color);
515
516 setPiece(promotionPiece, toSquare);
517 } else {
518 setPiece(movedPiece, toSquare);
519 }
520
521 if (moveType == EN_PASSANT) {
522 if (sideToMove == WHITE) {
523 removePiece(BLACK_PAWN, toSquare + SOUTH);
524 } else {
525 removePiece(WHITE_PAWN, toSquare + NORTH);
526 }
527 } else if (moveType == CASTLING) {
528 if (toSquare == G1) {
529 removePiece(WHITE_ROOK, H1);
530 setPiece(WHITE_ROOK, F1);
531 } else if (toSquare == C1) {
532 removePiece(WHITE_ROOK, A1);
533 setPiece(WHITE_ROOK, D1);
534 } else if (toSquare == G8) {
535 removePiece(BLACK_ROOK, H8);
536 setPiece(BLACK_ROOK, F8);
537 } else if (toSquare == C8) {
538 removePiece(BLACK_ROOK, A8);
539 setPiece(BLACK_ROOK, D8);
540 }
541
542 if (sideToMove == WHITE) {
543 if (castlingRights & WHITE_KINGSIDE) {
544 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_KINGSIDE_INDEX);
545 }
546
547 if (castlingRights & WHITE_QUEENSIDE) {
548 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_QUEENSIDE_INDEX);
549 }
550
551 castlingRights &= ~WHITE_CASTLING;
552 } else {
553 if (castlingRights & BLACK_KINGSIDE) {
554 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_KINGSIDE_INDEX);
555 }
556
557 if (castlingRights & BLACK_QUEENSIDE) {
558 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_QUEENSIDE_INDEX);
559 }
560
561 castlingRights &= ~BLACK_CASTLING;
562 }
563 }
564
565 if (movedPiece == WHITE_KING) {
566 if (castlingRights & WHITE_KINGSIDE) {
567 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_KINGSIDE_INDEX);
568 }
569
570 if (castlingRights & WHITE_QUEENSIDE) {
571 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_QUEENSIDE_INDEX);
572 }
573
574 castlingRights &= ~WHITE_CASTLING;
575 } else if (movedPiece == BLACK_KING) {
576 if (castlingRights & BLACK_KINGSIDE) {
577 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_KINGSIDE_INDEX);
578 }
579
580 if (castlingRights & BLACK_QUEENSIDE) {
581 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_QUEENSIDE_INDEX);
582 }
583
584 castlingRights &= ~BLACK_CASTLING;
585 } else if (movedPiece == WHITE_ROOK) {
586 if (fromSquare == A1) {
587 if (castlingRights & WHITE_QUEENSIDE) {
588 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_QUEENSIDE_INDEX);
589 castlingRights &= ~WHITE_QUEENSIDE;
590 }
591 } else if (fromSquare == H1) {
592 if (castlingRights & WHITE_KINGSIDE) {
593 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_KINGSIDE_INDEX);
594 castlingRights &= ~WHITE_KINGSIDE;
595 }
596 }
597 } else if (movedPiece == BLACK_ROOK) {
598 if (fromSquare == A8) {
599 if (castlingRights & BLACK_QUEENSIDE) {
600 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_QUEENSIDE_INDEX);
601 castlingRights &= ~BLACK_QUEENSIDE;
602 }
603 } else if (fromSquare == H8) {
604 if (castlingRights & BLACK_KINGSIDE) {
605 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_KINGSIDE_INDEX);
606 castlingRights &= ~BLACK_KINGSIDE;
607 }
608 }
609 }
610
611 if (movedPieceType == PAWN) {
612 halfMoveClock = 0;
613
614 if ((fromSquare ^ toSquare) == 16) {
615 if (sideToMove == WHITE) {
616 enPassantSquare = toSquare + SOUTH;
617 zobristHash ^= getZobristConstant(ZOBRIST_EN_PASSANT_START_INDEX + (enPassantSquare % 8));
618 } else {
619 enPassantSquare = toSquare + NORTH;
620 zobristHash ^= getZobristConstant(ZOBRIST_EN_PASSANT_START_INDEX + (enPassantSquare % 8));
621 }
622 }
623 }
624
625 if (sideToMove == BLACK) {
626 fullmoveClock += 1;
627 }
628
629 previousMove = move;
630 sideToMove = !sideToMove;
631 zobristHash ^= getZobristConstant(ZOBRIST_SIDE_TO_MOVE_INDEX);
632
633 assert(ply < MAX_PLIES);
634 ply++;
635 assert(ply < MAX_PLIES);
636 assert(enPassantSquare == NO_EN_PASSANT || (enPassantSquare / 8 == 2 || enPassantSquare / 8 == 5));
637}

642void Board::unmakeMove() {
643 ply--;
644 assert(ply >= 0 && ply < MAX_PLIES);
645 const BoardState& state = history[ply];
646 const Square fromSquare = getFromSquare(state.move);
647 const Square toSquare = getToSquare(state.move);
648 const MoveType moveType = getMoveType(state.move);
649 Piece movedPiece = getPieceOnSquare(toSquare);
650 const PieceColor movedColor = getPieceColor(movedPiece);
651
652 removePiece(movedPiece, toSquare);
653
654 if (moveType == PROMOTION) {
655 const PieceColor color = getPieceColor(movedPiece);
656 movedPiece = static_cast<Piece>(WHITE_PAWN + color);
657 }
658
659 setPiece(movedPiece, fromSquare);
660
661 if (state.capturedPiece != EMPTY) {
662 Square capturedSquare = toSquare;
663
664 if (moveType == EN_PASSANT) {
665 capturedSquare =
666 movedColor == WHITE ? static_cast<Square>(toSquare + SOUTH) : static_cast<Square>(toSquare + NORTH);
667 }
668
669 setPiece(state.capturedPiece, capturedSquare);
670 }
671
672 if (moveType == CASTLING) {
673 if (toSquare == G1) {
674 removePiece(WHITE_ROOK, F1);
675 setPiece(WHITE_ROOK, H1);
676 } else if (toSquare == C1) {
677 removePiece(WHITE_ROOK, D1);
678 setPiece(WHITE_ROOK, A1);
679 } else if (toSquare == G8) {
680 removePiece(BLACK_ROOK, F8);

701void Board::makeNullMove() {
702 history[ply].move = NO_MOVE;
703 history[ply].previousMove = previousMove;
704 history[ply].capturedPiece = EMPTY;
705 history[ply].enPassantSquare = enPassantSquare;
706 history[ply].castlingRights = castlingRights;
707 history[ply].zobristHash = zobristHash;
708 history[ply].halfMoveClock = halfMoveClock;
709
710 if (enPassantSquare != NO_EN_PASSANT) {
711 zobristHash ^= getZobristConstant(ZOBRIST_EN_PASSANT_START_INDEX + (enPassantSquare % 8));
712 }
713
714 enPassantSquare = NO_EN_PASSANT;
715
716 halfMoveClock += 1;
717
718 if (sideToMove == BLACK) {
719 fullmoveClock += 1;
720 }
721
722 previousMove = NO_MOVE;
723 sideToMove = !sideToMove;
724 zobristHash ^= getZobristConstant(ZOBRIST_SIDE_TO_MOVE_INDEX);
725
726 assert(ply < MAX_PLIES);
727 ply++;
728 assert(ply < MAX_PLIES);
729}

805bool Board::setFromFEN(const std::string_view fen) {
806 uint8_t index = A8;
807 int spaces = 0;
808
809 reset();
810
811 for (const char character : fen) {
812 if (character == ' ') {
813 spaces++;
814 continue;
815 }
816
817 if (character == ',') {
818 break;
819 }
820
821 if (spaces == 0) {
822 if (character == '/') {
823 index -= 16;
824 continue;
825 }
826
827 if (character >= '1' && character <= '8') {
828 index += character - '0';
829 continue;
830 }
831
832 if (character >= 'A' && character <= 'z') {
833 setPieceFromFENChar(character, index);
834 index++;
835 } else {
836 return false;
837 }
838 }
839
840 if (spaces == 1) {
841 if (tolower(character) == 'w') {
842 sideToMove = WHITE;
843 } else if (tolower(character) == 'b') {
844 sideToMove = BLACK;
845 zobristHash ^= getZobristConstant(ZOBRIST_SIDE_TO_MOVE_INDEX);
846 } else {
847 return false;
848 }
849 }
850
851 if (spaces == 2) {
852 if (character == '-') {
853 continue;
854 } else if (character == 'K') {
855 castlingRights |= WHITE_KINGSIDE;
856 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_KINGSIDE_INDEX);
857 continue;
858 }
859
860 if (character == 'Q') {
861 castlingRights |= WHITE_QUEENSIDE;
862 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_WHITE_QUEENSIDE_INDEX);
863 continue;
864 }
865
866 if (character == 'k') {
867 castlingRights |= BLACK_KINGSIDE;
868 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_KINGSIDE_INDEX);
869 continue;
870 }
871
872 if (character == 'q') {
873 castlingRights |= BLACK_QUEENSIDE;
874 zobristHash ^= getZobristConstant(ZOBRIST_CASTLING_BLACK_QUEENSIDE_INDEX);
875 continue;
876 }
877
878 return false;
879 }
880
881 if (spaces == 3) {
882 if (character == '-') {
883 continue;
884 }
885
886 if (tolower(character) < 'a' || tolower(character) > 'h') {
887 continue;
888 }
889
890 const int8_t file = tolower(character) - 'a';
891 const int8_t rank = (!sideToMove) == WHITE ? 2 : 5;
892
893 if (file < 0 || file > 7) {
894 return false;
895 }
896
897 enPassantSquare = rank * 8 + file;
898 zobristHash ^= getZobristConstant(ZOBRIST_EN_PASSANT_START_INDEX + (enPassantSquare % 8));
899 index += 2;
900 }
901
902 if (spaces == 4) {
903 if (character >= '0' && character <= '9') {
904 halfMoveClock = halfMoveClock * 10 + (character - '0');
905 }
906 }
907
908 if (spaces == 5) {
909 if (character >= '0' && character <= '9') {
910 fullmoveClock = fullmoveClock * 10 + (character - '0');
911 }
912 }
913 }
914
915 return true;
916}
