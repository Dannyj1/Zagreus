36bool MovePicker::next(Move& move) {
37 if (currentIndex >= moveList.size) {
38 return false;
39 }
40
41 int bestIndex = currentIndex;
42 int bestScore = scores[currentIndex];
43
44 for (int i = currentIndex + 1; i < moveList.size; ++i) {
45 if (scores[i] > bestScore) {
46 bestIndex = i;
47 bestScore = scores[i];
48 }
49 }
50
51 std::swap(scores[currentIndex], scores[bestIndex]);
52 std::swap(moveList.moves[currentIndex], moveList.moves[bestIndex]);
53
54 move = moveList.moves[currentIndex];
55 assert(move != NO_MOVE);
56
57 if (move == NO_MOVE) {
58 return false;
59 }
60
61 currentIndex += 1;
62 return true;
63}

72void MovePicker::score(Board& board) {
73 const PvLine& pvLine = board.getPreviousPvLine();
74 const int currentPly = board.getPly();
75 const int pvMoveIndex = currentPly - pvLine.startPly;
76 Move pvMove = NO_MOVE;
77 Move ttMove = NO_MOVE;
78
79 if (pvLine.moveCount > pvMoveIndex) {
80 assert(pvLine.moves[pvMoveIndex] != NO_MOVE);
81 pvMove = pvLine.moves[pvMoveIndex];
82 }
83
84 const uint64_t zobristHash = board.getZobristHash();
85 const TTEntry* entry = tt->getEntry(zobristHash);
86
87 if (entry != nullptr) {
88 ttMove = entry->bestMove;
89
90 if (ttMove == pvMove) {
91 ttMove = NO_MOVE;
92 }
93 }
94
95 // const Move killerMove1 = tt->getKillerMove(currentPly, 0);
96 // const Move killerMove2 = tt->getKillerMove(currentPly, 1);
97
98 for (int i = 0; i < moveList.size; ++i) {
99 const Move move = moveList.moves[i];
100 const MoveType moveType = getMoveType(move);
101 const Square toSquare = getToSquare(move);
102 Piece capturedPiece = board.getPieceOnSquare(toSquare);
103
104 if (moveType == EN_PASSANT) {
105 capturedPiece = board.getSideToMove() == WHITE ? BLACK_PAWN : WHITE_PAWN;
106 }
107
108 if (move == pvMove) {
109 scores[i] = 5000000;
110 } else if (move == ttMove) {
111 scores[i] = 2500000;
112 } else if (capturedPiece != EMPTY) {
113 // SEE for identifying good and bad captures, then further ordering by capture history
114 const Piece movingPiece = board.getPieceOnSquare(getFromSquare(move));
115 const int captureHistoryValue = tt->getCaptureHistoryValue(move, movingPiece, capturedPiece);
116 const int captureScore = (getPieceValue(capturedPiece) * 10) + captureHistoryValue;
117 const bool isGoodCapture = board.see(move, 0);
118 const int seeScore = isGoodCapture ? 1500000 + captureScore : -1500000 + captureScore;
119
120 scores[i] = seeScore;
121 } else if (moveType == PROMOTION) {
122 const Piece promotedPiece = getPieceFromPromotionPiece(getPromotionPiece(move), board.getSideToMove());
123 const int promotionScore = 1500000 + getPieceValue(promotedPiece);
124
125 scores[i] = promotionScore;
126 // } else if (move == killerMove1) {
127 // scores[i] = 900000;
128 // } else if (move == killerMove2) {
129 // scores[i] = 800000;
130 } else {
131 // Ordering of quiet moves
132 const int historyValue = tt->getHistoryValue(board.getSideToMove(), move);
133
134 scores[i] = historyValue;
135 }
136 }
137}
