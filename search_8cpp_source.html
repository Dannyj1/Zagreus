43void initializeSearch() {
44 for (int depth = 1; depth < MAX_PLIES; ++depth) {
45 for (int moveCount = 1; moveCount < MAX_MOVES; ++moveCount) {
46 lmrTable[depth][moveCount] = static_cast<int>(0.5 + 0.5 * std::log(depth) * std::log(moveCount));
47 }
48 }
49}

51template <PieceColor color>
52Move search(Engine& engine, Board& board, SearchParams& params, SearchStats& stats) {
53 int depth = 1;
54 const int currentPly = board.getPly();
55 int searchTime;
56 auto endTime = std::chrono::steady_clock::time_point{};
57
58 if (params.infinite) {
59 searchTime = 0;
60 } else {
61 searchTime = calculateSearchTime<color>(params);
62 endTime = std::chrono::steady_clock::now() + std::chrono::milliseconds(searchTime);
63 }
64
65 const auto startTime = std::chrono::steady_clock::now();
66 PvLine bestPvLine = PvLine{board.getPly()};
67
68 engine.setSearchStopped(false);
69
70 while (!engine.isSearchStopped() && (currentPly + depth) < MAX_PLIES) {
71 if (!params.infinite && (params.blackTime > 0 || params.whiteTime > 0)) {
72 // Don't start the next iteration if we are 10% away from the end time
73 if (std::chrono::steady_clock::now() + std::chrono::milliseconds(searchTime / 10) > endTime) {
74 engine.setSearchStopped(true);
75 break;
76 }
77 }
78
79 if (params.depth > 0 && depth > params.depth) {
80 engine.setSearchStopped(true);
81 break;
82 }
83
84 if (params.max_nodes > 0 && stats.nodesSearched + stats.qNodesSearched >= params.max_nodes) {
85 engine.setSearchStopped(true);
86 break;
87 }
88
89 PvLine pvLine = PvLine{board.getPly()};
90
91 const int score =
92 pvSearch<color, ROOT>(engine, board, INITIAL_ALPHA, INITIAL_BETA, depth, stats, endTime, pvLine);
93 assert(score != INITIAL_ALPHA && score != INITIAL_BETA);
94 assert(depth > 0);
95
96 if (endTime.time_since_epoch().count() != 0 && std::chrono::steady_clock::now() > endTime) {
97 engine.setSearchStopped(true);
98 break;
99 }
100
101 if (engine.isSearchStopped()) {
102 break;
103 }
104
105 bestPvLine = pvLine;
106 board.setPreviousPvLine(bestPvLine);
107
108 stats.score = score;
109 stats.timeSpentMs =
110 std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - startTime).count();
111 stats.depth = depth;
112
113 // Make sure we don't divide by zero
114 if (stats.timeSpentMs == 0) {
115 stats.timeSpentMs = 1;
116 }
117
118 const uint64_t totalNodesSearch = stats.nodesSearched + stats.qNodesSearched;
119 const uint64_t nps = static_cast<double>(totalNodesSearch) / (static_cast<double>(stats.timeSpentMs) / 1000.0);
120 std::string pvString = parsePvLine(bestPvLine);
121 engine.sendInfoMessage("depth " + std::to_string(stats.depth) + " score cp " + std::to_string(stats.score) +
122 " nodes " + std::to_string(totalNodesSearch) + " time " +
123 std::to_string(stats.timeSpentMs) + " nps " + std::to_string(nps) + " pv " + pvString);
124 depth += 1;
125 }
126
127 if (bestPvLine.moves[0] == NO_MOVE) {
128 // Find the first legal move and play that
129 MoveList moves = MoveList{};
130 generateMoves<color, ALL>(board, moves);
131 MovePicker movePicker{moves};
132 Move move;
133 Move bestMove = NO_MOVE;
134
135 while (movePicker.next(move)) {
136 board.makeMove(move);
137
138 if (!board.isPositionLegal<color>()) {
139 board.unmakeMove();
140 continue;
141 }
142
143 bestMove = move;
144 board.unmakeMove();
145 break;
146 }
147
148 assert(bestMove != NO_MOVE);
149 return bestMove;
150 }
151
152 assert(bestPvLine.moves[0] != NO_MOVE);
153 board.setPreviousPvLine(bestPvLine);
154 return bestPvLine.moves[0];
155}

160template <PieceColor color, NodeType nodeType>
161int pvSearch(Engine& engine, Board& board, int alpha, int beta, int depth, SearchStats& stats,
162 const std::chrono::time_point<std::chrono::steady_clock>& endTime, PvLine& pvLine) {
163 constexpr bool isPV = nodeType == PV || nodeType == ROOT;
164 constexpr bool isRoot = nodeType == ROOT;
165 constexpr PieceColor opponentColor = !color;
166
167 if (board.isDraw()) {
168 return DRAW_SCORE;
169 }
170
171 if (!isRoot && (stats.nodesSearched + stats.qNodesSearched) % 4096 == 0) {
172 if (endTime.time_since_epoch().count() != 0 && std::chrono::steady_clock::now() > endTime) {
173 engine.setSearchStopped(true);
174 return beta;
175 }
176 }
177
178 bool isInCheck = board.isKingInCheck<color>();
179
180 if (isInCheck) {
181 depth += 1;
182#ifdef TRACE_SEARCH
183 stats.checkExtensions++;
184#endif
185 }
186
187 if (depth <= 0) {
188 assert(!isRoot);
189 pvLine.moveCount = 0;
190 return qSearch<color, nodeType>(engine, board, alpha, beta, depth, stats, endTime);
191 }
192
193 stats.nodesSearched += 1;
194 const int eval = Evaluation(board).evaluate();
195
196 if (!isPV) {
197 const int16_t score = tt->probePosition(board.getZobristHash(), depth, alpha, beta, board.getPly());
198
199#ifdef TRACE_SEARCH
200 stats.ttProbes++;
201#endif
202
203 if (score != NO_TT_SCORE) {
204#ifdef TRACE_SEARCH
205 stats.ttHits++;
206#endif
207 return score;
208 }
209
210 // Reverse Futility Pruning
211 int rfMargin = 150 * depth;
212 if (!isInCheck && depth <= 3 && eval >= beta + rfMargin) {
213#ifdef TRACE_SEARCH
214 stats.reverseFutilityPrunes++;
215#endif
216 return eval;
217 }
218
219 // Null Move Pruning
220 if (depth >= 2 && !isInCheck && board.hasNonPawnMaterial<color>() && board.getPreviousMove() != NO_MOVE) {
221#ifdef TRACE_SEARCH
222 stats.nmpTries++;
223#endif
224 board.makeNullMove();
225 const int R = 2 + depth / 3;
226 PvLine nmpPvLine = PvLine{board.getPly()};
227 const int nullMoveScore = -pvSearch<opponentColor, REGULAR>(engine, board, -beta, -beta + 1, depth - R,
228 stats, endTime, nmpPvLine);
229 board.unmakeNullMove();
230
231 if (nullMoveScore >= beta) {
232#ifdef TRACE_SEARCH
233 stats.nmpPrunes++;
234#endif
235 return nullMoveScore;
236 }
237 }
238 }
239
240 bool firstMove = true;
241 int legalMoves = 0;
242
243 Move move;
244 MoveList moves = MoveList{};
245
246 if (isInCheck) {
247 generateMoves<color, EVASIONS>(board, moves);
248 } else {
249 generateMoves<color, ALL>(board, moves);
250 }
251
252 MoveList searchedQuietMoves{};
253 MoveList searchedCaptures{};
254 MovePicker movePicker{moves};
255 movePicker.score(board);
256
257 PvLine nodePvLine = PvLine{board.getPly()};
258 Move bestMove = NO_MOVE;
259 int bestScore = INT32_MIN;
260 int movesSearched = 0;
261
262 while (movePicker.next(move)) {
263 const MoveType moveType = getMoveType(move);
264 const Square toSquare = getToSquare(move);
265 Piece capturedPiece = board.getPieceOnSquare(toSquare);
266
267 if (moveType == EN_PASSANT) {
268 capturedPiece = color == WHITE ? BLACK_PAWN : WHITE_PAWN;
269 }
270
271 board.makeMove(move);
272
273 if (!board.isPositionLegal<color>()) {
274 board.unmakeMove();
275 continue;
276 }
277
278 legalMoves += 1;
279
280 // Futility pruning
281 if (depth <= 4 && !isInCheck && capturedPiece == EMPTY && getMoveType(move) != PROMOTION) {
282 const int futilityMargin = 200 * depth;
283 const int futilityScore = eval + futilityMargin;
284
285 if (futilityScore <= alpha) {
286#ifdef TRACE_SEARCH
287 stats.futilityPrunes++;
288#endif
289
290 if (futilityScore > bestScore) {
291 bestScore = futilityScore;
292 }
293
294 board.unmakeMove();
295 continue;
296 }
297 }
298
299 if (capturedPiece == EMPTY) {
300 searchedQuietMoves.moves[searchedQuietMoves.size++] = move;
301 } else {
302 searchedCaptures.moves[searchedCaptures.size++] = move;
303 }
304
305 bool doFullSearch = true;
306 int score = INT32_MIN;
307
308 // Late Move Reduction
309 if (movesSearched > 3 && depth >= 3 && capturedPiece == EMPTY && getMoveType(move) != PROMOTION) {
310 doFullSearch = false;
311 int R = 0;
312
313 R = lmrTable[depth][movesSearched];
314 R -= isPV;
315 R -= isInCheck;
316 R -= board.isKingInCheck<opponentColor>();
317
318 // Make sure depth - 1 - R is at least 1
319 if (depth - 1 - R <= 0) {
320 R = depth - 2;
321 }
322
323 R = std::max(0, R);
324
325#ifdef TRACE_SEARCH
326 stats.lmrSearches++;
327#endif
328 score = -pvSearch<opponentColor, REGULAR>(engine, board, -alpha - 1, -alpha, depth - 1 - R, stats, endTime,
329 nodePvLine);
330
331 if (score > alpha) {
332#ifdef TRACE_SEARCH
333 stats.lmrResearches++;
334#endif
335 doFullSearch = true;
336 }
337 }
338
339 if (doFullSearch) {
340 if (firstMove) {
341 if (isRoot) {
342 score = -pvSearch<opponentColor, PV>(engine, board, -beta, -alpha, depth - 1, stats, endTime,
343 nodePvLine);
344 } else {
345 score = -pvSearch<opponentColor, nodeType>(engine, board, -beta, -alpha, depth - 1, stats, endTime,
346 nodePvLine);
347 }
348
349 firstMove = false;
350 } else {
351 score = -pvSearch<opponentColor, REGULAR>(engine, board, -alpha - 1, -alpha, depth - 1, stats, endTime,
352 nodePvLine);
353
354 if (isPV && score > alpha && (isRoot || score < beta)) {
355 score = -pvSearch<opponentColor, PV>(engine, board, -beta, -alpha, depth - 1, stats, endTime,
356 nodePvLine);
357 }
358 }
359 }
360
361 board.unmakeMove();
362
363 if (score >= beta) {
364#ifdef TRACE_SEARCH
365 if (movesSearched == 0) {
366 stats.firstMoveCutoffs++;
367 } else if (movesSearched == 1) {
368 stats.secondMoveCutoffs++;
369 }
370 stats.totalMoveCutoffNumber += (movesSearched + 1);
371 stats.totalCutoffs++;
372#endif
373
374 if (!engine.isSearchStopped()) {
375 if (capturedPiece == EMPTY && getMoveType(move) != PROMOTION) {
376 // Killer move heuristic
377 tt->addKillerMove(move, board.getPly());
378
379 // History heuristic
380 const int historyValue = 300 * depth - 250;
381
382 tt->updateHistory<color>(move, historyValue);
383
384 for (int i = 0; i < searchedQuietMoves.size; ++i) {
385 const Move quietMove = searchedQuietMoves.moves[i];
386
387 if (quietMove != move) {
388 tt->updateHistory<color>(quietMove, -historyValue);
389 }
390 }
391 }
392
393 if (capturedPiece != EMPTY) {
394 const int historyValue = 300 * depth - 250;
395 const Piece movingPiece = board.getPieceOnSquare(getFromSquare(move));
396
397 tt->updateCaptureHistory(move, movingPiece, capturedPiece, historyValue);
398
399 for (int i = 0; i < searchedCaptures.size; ++i) {
400 const Move captureMove = searchedCaptures.moves[i];
401
402 if (captureMove != move) {
403 const Piece otherMovingPiece = board.getPieceOnSquare(getFromSquare(captureMove));
404 const MoveType otherMoveType = getMoveType(captureMove);
405 Piece otherCapturedPiece = board.getPieceOnSquare(getToSquare(captureMove));
406
407 if (otherMoveType == EN_PASSANT) {
408 otherCapturedPiece = color == WHITE ? BLACK_PAWN : WHITE_PAWN;
409 }
410
411 tt->updateCaptureHistory(captureMove, otherMovingPiece, otherCapturedPiece, -historyValue);
412 }
413 }
414 }
415
416 if (!isRoot) {
417#ifdef TRACE_SEARCH
418 stats.ttWrites++;
419#endif
420 tt->savePosition(board.getZobristHash(), depth, board.getPly(), score, move, BETA);
421 }
422 }
423
424 return score;
425 }
426
427 if (score > bestScore) {
428 bestScore = score;
429
430 if (score > alpha) {
431 bestMove = move;
432 alpha = score;
433 pvLine.moves[0] = move;
434 std::memcpy(pvLine.moves + 1, nodePvLine.moves, nodePvLine.moveCount * sizeof(Move));
435 pvLine.moveCount = nodePvLine.moveCount + 1;
436 }
437 }
438
439 movesSearched += 1;
440 }
441
442 if (!legalMoves) {
443 if (isInCheck) {
444 alpha = -MATE_SCORE + board.getPly();
445 bestScore = alpha;
446 } else {
447 alpha = DRAW_SCORE;
448 bestScore = DRAW_SCORE;
449 }
450 }
451
452 if (!isRoot) {
453 const TTNodeType ttNodeType = (isPV && bestMove != NO_MOVE) ? EXACT : ALPHA;
454
455 if (!engine.isSearchStopped()) {
456#ifdef TRACE_SEARCH
457 stats.ttWrites++;
458#endif
459 tt->savePosition(board.getZobristHash(), depth, board.getPly(), bestScore, bestMove, ttNodeType);
460 }
461 }
462
463 assert(bestScore != INITIAL_ALPHA);
464 return bestScore;
465}

467template <PieceColor color, NodeType nodeType>
468int qSearch(Engine& engine, Board& board, int alpha, int beta, int depth, SearchStats& stats,
469 const std::chrono::time_point<std::chrono::steady_clock>& endTime) {
470 assert(nodeType != ROOT);
471 constexpr bool isPV = nodeType == PV;
472
473 if (board.isDraw()) {
474 return DRAW_SCORE;
475 }
476
477 if ((stats.nodesSearched + stats.qNodesSearched) % 4096 == 0) {
478 if (endTime.time_since_epoch().count() != 0 && std::chrono::steady_clock::now() > endTime) {
479 engine.setSearchStopped(true);
480 return beta;
481 }
482 }
483
484 if (!isPV) {
485 const int16_t score = tt->probePosition(board.getZobristHash(), depth, alpha, beta, board.getPly());
486
487#ifdef TRACE_SEARCH
488 stats.qTtProbes++;
489#endif
490
491 if (score != NO_TT_SCORE) {
492#ifdef TRACE_SEARCH
493 stats.qTtHits++;
494#endif
495 return score;
496 }
497 }
498
499 stats.qNodesSearched += 1;
500
501 const bool isInCheck = board.isKingInCheck<color>();
502 int bestScore;
503
504 if (isInCheck) {
505 bestScore = -MATE_SCORE;
506 } else {
507 bestScore = Evaluation(board).evaluate();
508
509 // Stand pat
510 if (bestScore >= beta) {
511 if (!engine.isSearchStopped()) {
512#ifdef TRACE_SEARCH
513 stats.ttWrites++;
514#endif
515 tt->savePosition(board.getZobristHash(), depth, board.getPly(), bestScore, NO_MOVE, BETA);
516 }
517
518 return bestScore;
519 }
520
521 if (bestScore > alpha) {
522 alpha = bestScore;
523 }
524 }
525
526 int legalMoves = 0;
527 Move move;
528 MoveList moves = MoveList{};
529 Move bestMove = NO_MOVE;
530
531 if (isInCheck) {
532 generateMoves<color, EVASIONS>(board, moves);
533 } else {
534 generateMoves<color, QSEARCH>(board, moves);
535 }
536
537 MovePicker movePicker{moves};
538 movePicker.score(board);
539
540 while (movePicker.next(move)) {
541 const MoveType moveType = getMoveType(move);
542 const Square toSquare = getToSquare(move);
543 Piece capturedPiece = board.getPieceOnSquare(toSquare);
544
545 if (moveType == EN_PASSANT) {
546 capturedPiece = color == WHITE ? BLACK_PAWN : WHITE_PAWN;
547 }
548
549 if (!isInCheck && capturedPiece != EMPTY) {
550 const bool see = board.see(move, 0);
551
552 if (!see) {
553#ifdef TRACE_SEARCH
554 stats.seePrunes++;
555#endif
556 continue;
557 }
558 }
559
560 board.makeMove(move);
561
562 if (!board.isPositionLegal<color>()) {
563 board.unmakeMove();
564 continue;
565 }
566
567 legalMoves += 1;
568
569 const int score = -qSearch<!color, nodeType>(engine, board, -beta, -alpha, depth - 1, stats, endTime);
570
571 board.unmakeMove();
572
573 if (score >= beta) {
574 if (!engine.isSearchStopped()) {
575#ifdef TRACE_SEARCH
576 stats.ttWrites++;
577#endif
578 tt->savePosition(board.getZobristHash(), depth, board.getPly(), score, move, BETA);
579 }
580
581 return score;
582 }
583
584 if (score > bestScore) {
585 bestScore = score;
586
587 if (score > alpha) {
588 bestMove = move;
589 alpha = score;
590 }
591 }
592 }
593
594 if (!legalMoves && isInCheck) {
595 bestScore = -MATE_SCORE + board.getPly();
596 }
597
598 const TTNodeType ttNodeType = (isPV && bestMove != NO_MOVE) ? EXACT : ALPHA;
599
600 if (!engine.isSearchStopped()) {
601#ifdef TRACE_SEARCH
602 stats.ttWrites++;
603#endif
604 tt->savePosition(board.getZobristHash(), depth, board.getPly(), bestScore, bestMove, ttNodeType);
605 }
606
607 assert(bestScore != INITIAL_ALPHA);
608 return bestScore;
609}

636void SearchStats::printTrace(Engine& engine, int numPositions) const {
637 const uint64_t totalNodes = nodesSearched + qNodesSearched;
638 const uint64_t avgTotalNodes = totalNodes / numPositions;
639 const double branchFactor = avgTotalNodes > 1 && depth > 1 ? std::pow(avgTotalNodes, 1.0 / depth) : 0.0;
640
641 const double ttHitRate = ttProbes > 0 ? static_cast<double>(ttHits) / ttProbes * 100.0 : 0.0;
642 const double qTtHitRate = qTtProbes > 0 ? static_cast<double>(qTtHits) / qTtProbes * 100.0 : 0.0;
643 const double nmpSuccessRate = nmpTries > 0 ? static_cast<double>(nmpPrunes) / nmpTries * 100.0 : 0.0;
644 const double lmrResearchRate = lmrSearches > 0 ? static_cast<double>(lmrResearches) / lmrSearches * 100.0 : 0.0;
645 const double firstMoveCutoffRate =
646 totalCutoffs > 0 ? static_cast<double>(firstMoveCutoffs) / totalCutoffs * 100.0 : 0.0;
647 const double secondMoveCutoffRate =
648 totalCutoffs > 0 ? static_cast<double>(secondMoveCutoffs) / totalCutoffs * 100.0 : 0.0;
649 const double avgMoveCutoff = totalCutoffs > 0 ? static_cast<double>(totalMoveCutoffNumber) / totalCutoffs : 0.0;
650
651 std::stringstream ss;
652 ss << "\n--- Search Statistics" << (numPositions > 1 ? " (Averages per position)" : "") << " ---\n"
653 << "Effective Branching Factor: " << std::fixed << std::setprecision(2) << branchFactor << "\n"
654 << "\n--- Transposition Table ---\n"
655 << "Probes: " << ttProbes / numPositions << "\n"
656 << "Hits: " << ttHits / numPositions << " (" << ttHitRate << "%)\n"
657 << "Writes: " << ttWrites / numPositions << "\n"
658 << "Q-Search Probes: " << qTtProbes / numPositions << "\n"
659 << "Q-Search Hits: " << qTtHits / numPositions << " (" << qTtHitRate << "%)\n"
660 << "\n--- Move Ordering ---\n"
661 << "First Move Cutoffs: " << firstMoveCutoffs / numPositions << " (" << firstMoveCutoffRate << "%)\n"
662 << "Second Move Cutoffs: " << secondMoveCutoffs / numPositions << " (" << secondMoveCutoffRate << "%)\n"
663 << "Average Move Cutoff: " << avgMoveCutoff << "\n"
664 << "\n--- Pruning, Reductions, and Extensions ---\n"
665 << "NMP Tries: " << nmpTries / numPositions << "\n"
666 << "NMP Prunes: " << nmpPrunes / numPositions << " (" << nmpSuccessRate << "%)\n"
667 << "LMR Searches: " << lmrSearches / numPositions << "\n"
668 << "LMR Researches: " << lmrResearches / numPositions << " (" << lmrResearchRate << "%)\n"
669 << "Futility Prunes: " << futilityPrunes / numPositions << "\n"
670 << "Reverse Futility Prunes: " << reverseFutilityPrunes / numPositions << "\n"
671 << "Check Extensions: " << checkExtensions / numPositions << "\n"
672 << "\n--- Quiescence Search ---\n"
673 << "SEE Prunes: " << seePrunes / numPositions << "\n"
674 << "-------------------------";
675 engine.sendInfoMessage(ss.str());
676}
