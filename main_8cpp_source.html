171void benchmark(bool fast) {
172 Engine engine{};
173 uint64_t nodes = 0;
174 double totalMs = 0;
175 Board board{};
176
177#ifdef TRACE_SEARCH
178 SearchStats totalStats{};
179 totalStats.clearTrace();
180#endif
181
182 engine.registerOptions();
183 engine.doSetup();
184 TranspositionTable::getTT()->setTableSize(128);
185 std::vector<std::string> positions = fast ? FAST_BENCHMARK_POSITIONS : BENCHMARK_POSITIONS;
186 SearchParams params{};
187
188 params.depth = fast ? 5 : 10;
189
190 for (const std::string& position : positions) {
191 board.setFromFEN(position);
192 const PieceColor color = board.getSideToMove();
193
194 TranspositionTable::getTT()->reset();
195 SearchStats stats{};
196#ifdef TRACE_SEARCH
197 stats.clearTrace();
198#endif
199 auto start = std::chrono::steady_clock::now();
200
201 if (color == WHITE) {
202 search<WHITE>(engine, board, params, stats);
203 } else {
204 search<BLACK>(engine, board, params, stats);
205 }
206
207 auto end = std::chrono::steady_clock::now();
208 std::chrono::duration<double, std::milli> elapsed = end - start;
209
210 nodes += stats.nodesSearched + stats.qNodesSearched;
211 totalMs += elapsed.count();
212
213#ifdef TRACE_SEARCH
214 totalStats += stats;
215#endif
216 }
217
218#ifdef TRACE_SEARCH
219 totalStats.depth = params.depth;
220 totalStats.printTrace(engine, positions.size());
221#endif
222
223 if (nodes == 0 || totalMs == 0) {
224 engine.sendMessage("0 nodes 0 nps");
225 return;
226 }
227
228 const double secondsSpent = totalMs / 1000.0;
229 const uint64_t nodesPerSecond = static_cast<uint64_t>(static_cast<double>(nodes) / secondsSpent);
230 std::string message = std::to_string(nodes) + " nodes " + std::to_string(nodesPerSecond) + " nps";
231
232 engine.sendMessage(message);
233}
