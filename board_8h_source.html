209 template <Piece piece>
210 void setPiece(const uint8_t square) {
211 assert(piece != Piece::EMPTY);
212 assert(board[square] == Piece::EMPTY);
213 const uint64_t squareBB = squareToBitboard(square);
214
215 board[square] = piece;
216 bitboards[piece] |= squareBB;
217 occupied |= squareBB;
218 colorBoards[getPieceColor(piece)] |= squareBB;
219
220 const int zobristIndex = ZOBRIST_PIECE_START_INDEX + static_cast<int>(piece) * SQUARES + square;
221 zobristHash ^= getZobristConstant(zobristIndex);
222 }

518 template <PieceColor color>
519 [[nodiscard]] uint64_t backwardPawns() const {
520 if (color == WHITE) {
521 const uint64_t pawns = getPieceBoard<WHITE_PAWN>();
522 const uint64_t opponentPawns = getPieceBoard<BLACK_PAWN>();
523 const uint64_t stops = pawns << 8;
524 const uint64_t whiteAttackSpans = pawnAttackFrontSpans<WHITE>();
525 const uint64_t blackPawnAttacks = calculateBlackPawnAttacks(opponentPawns);
526
527 return (stops & blackPawnAttacks & ~whiteAttackSpans) >> 8;
528 } else {
529 const uint64_t pawns = getPieceBoard<BLACK_PAWN>();
530 const uint64_t opponentPawns = getPieceBoard<WHITE_PAWN>();
531 const uint64_t stops = pawns >> 8;
532 const uint64_t blackAttackSpans = pawnAttackFrontSpans<BLACK>();
533 const uint64_t whitePawnAttacks = calculateWhitePawnAttacks(opponentPawns);
534
535 return (stops & whitePawnAttacks & ~blackAttackSpans) << 8;
536 }
