196 template <Piece piece>
197 void setPiece(const uint8_t square) {
198 assert(piece != Piece::EMPTY);
199 assert(board[square] == Piece::EMPTY);
200 const uint64_t squareBB = squareToBitboard(square);
201
202 board[square] = piece;
203 bitboards[piece] |= squareBB;
204 occupied |= squareBB;
205 colorBoards[getPieceColor(piece)] |= squareBB;
206
207 const int zobristIndex = ZOBRIST_PIECE_START_INDEX + static_cast<int>(piece) * SQUARES + square;
208 zobristHash ^= getZobristConstant(zobristIndex);
209 }

514 template <PieceColor color>
515 [[nodiscard]] uint64_t backwardPawns() const {
516 if (color == WHITE) {
517 const uint64_t pawns = getPieceBoard<WHITE_PAWN>();
518 const uint64_t opponentPawns = getPieceBoard<BLACK_PAWN>();
519 const uint64_t stops = pawns << 8;
520 const uint64_t whiteAttackSpans = pawnAttackFrontSpans<WHITE>();
521 const uint64_t blackPawnAttacks = calculateBlackPawnAttacks(opponentPawns);
522
523 return (stops & blackPawnAttacks & ~whiteAttackSpans) >> 8;
524 } else {
525 const uint64_t pawns = getPieceBoard<BLACK_PAWN>();
526 const uint64_t opponentPawns = getPieceBoard<WHITE_PAWN>();
527 const uint64_t stops = pawns >> 8;
528 const uint64_t blackAttackSpans = pawnAttackFrontSpans<BLACK>();
529 const uint64_t whitePawnAttacks = calculateWhitePawnAttacks(opponentPawns);
530
531 return (stops & whitePawnAttacks & ~blackAttackSpans) << 8;
532 }

565 const uint64_t candidates = pawns & ~opponentFrontSpans;
566 return candidates & ~passedPawns<BLACK>();

570 template <PieceColor color>
571 [[nodiscard]] uint64_t passedPawns() const {
572 if (color == WHITE) {
573 const uint64_t whitePawns = getPieceBoard<WHITE_PAWN>();
574 const uint64_t blackFrontSpans = pawnFrontSpans<BLACK>();
575 const uint64_t blackAttackSpans = shiftEast(blackFrontSpans) | shiftWest(blackFrontSpans);
576 const uint64_t combinedSpans = blackFrontSpans | blackAttackSpans;
577
578 return whitePawns & ~combinedSpans;
579 } else {
580 const uint64_t blackPawns = getPieceBoard<BLACK_PAWN>();
581 const uint64_t whiteFrontSpans = pawnFrontSpans<WHITE>();
582 const uint64_t whiteAttackSpans = shiftEast(whiteFrontSpans) | shiftWest(whiteFrontSpans);
583 const uint64_t combinedSpans = whiteFrontSpans | whiteAttackSpans;
584
585 return blackPawns & ~combinedSpans;
