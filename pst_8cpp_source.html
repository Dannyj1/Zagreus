156void initializePst() {
157 for (Piece piece = WHITE_PAWN; piece <= BLACK_KING; piece++) {
158 for (Square square = A1; square <= H8; square++) {
159 const int midgamePieceValue = evalMaterialValues[MIDGAME][getPieceType(piece)];
160 const int endgamePieceValue = evalMaterialValues[ENDGAME][getPieceType(piece)];
161 const PieceColor color = getPieceColor(piece);
162
163 if (color == WHITE) {
164 midgamePstTable[piece][square] = midgamePieceValue + getMidgameTable(getPieceType(piece))[square ^ 56];
165 endgamePstTable[piece][square] = endgamePieceValue + getEndgameTable(getPieceType(piece))[square ^ 56];
166 } else {
167 midgamePstTable[piece][square] = midgamePieceValue + getMidgameTable(getPieceType(piece))[square];
168 endgamePstTable[piece][square] = endgamePieceValue + getEndgameTable(getPieceType(piece))[square];
169 }
170 }
171 }
