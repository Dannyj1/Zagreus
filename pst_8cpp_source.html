148void initializePst() {
149 for (Piece piece = WHITE_PAWN; piece <= BLACK_KING; piece++) {
150 for (Square square = A1; square <= H8; square++) {
151 const int midgamePieceValue = evalMaterialValues[MIDGAME][getPieceType(piece)];
152 const int endgamePieceValue = evalMaterialValues[ENDGAME][getPieceType(piece)];
153 const PieceColor color = getPieceColor(piece);
154
155 if (color == WHITE) {
156 midgamePstTable[piece][square] = midgamePieceValue + getMidgameTable(getPieceType(piece))[square ^ 56];
157 endgamePstTable[piece][square] = endgamePieceValue + getEndgameTable(getPieceType(piece))[square ^ 56];
158 } else {
159 midgamePstTable[piece][square] = midgamePieceValue + getMidgameTable(getPieceType(piece))[square];
160 endgamePstTable[piece][square] = endgamePieceValue + getEndgameTable(getPieceType(piece))[square];
161 }
162 }
163 }
