41void initializeAttackLookupTables() {
42 for (uint8_t square = 0; square < SQUARES; ++square) {
43 const uint64_t bb = squareToBitboard(square);

45 pawnAttacksTable[WHITE][square] = calculateWhitePawnAttacks(bb);
46 pawnAttacksTable[BLACK][square] = calculateBlackPawnAttacks(bb);
47 knightAttacksTable[square] = calculateKnightAttacks(bb);
48 kingAttacksTable[square] = calculateKingAttacks(bb);

55void initializeLookupTables() {
56 // Between lookup table
57 for (int from = 0; from < 64; from++) {
58 for (int to = 0; to < 64; to++) {
59 const uint64_t m1 = -1ULL;
60 const uint64_t a2a7 = 0x0001010101010100ULL;
61 const uint64_t b2g7 = 0x0040201008040200ULL;
62 const uint64_t h1b7 = 0x0002040810204080ULL;
63 uint64_t btwn, line, rank, file;
64
65 btwn = m1 << from ^ m1 << to;
66 file = (to & 7) - (from & 7);
67 rank = ((to | 7) - from) >> 3;
68 line = (file & 7) - 1 & a2a7;
69 line += 2 * (((rank & 7) - 1) >> 58);
70 line += (rank - file & 15) - 1 & b2g7;
71 line += (rank + file & 15) - 1 & h1b7;
72 line *= btwn & -btwn;
73
74 betweenLookupTable[from][to] = line & btwn;
75 }
76 }
77
78 // Neighboring files lookup table
79 for (uint8_t square = 0; square < SQUARES; ++square) {
80 const uint64_t fileMask = squareToFileMask(square);
81 const uint64_t neighboringFiles = shiftWest(fileMask) | shiftEast(fileMask);
82
83 neighboringFilesTable[square] = neighboringFiles;
84 }
85}

129uint64_t getBishopAttacks(const uint8_t square, uint64_t occupied) {
130 assert(square < SQUARES);
131 occupied &= getBishopMask(square);
132 occupied *= getBishopMagic(square);
133 occupied >>= 64 - BBits[square];
134
135 return getBishopMagicAttacks(square, occupied);
136}

144uint64_t getRookAttacks(const uint8_t square, uint64_t occupied) {
145 assert(square < SQUARES);

159uint64_t queenAttacks(const uint8_t square, const uint64_t occupied) {
