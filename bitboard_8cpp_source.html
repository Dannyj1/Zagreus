39void initializeAttackLookupTables() {
40 for (uint8_t square = 0; square < SQUARES; ++square) {
41 const uint64_t bb = squareToBitboard(square);
42
43 pawnAttacksTable[WHITE][square] = calculateWhitePawnAttacks(bb);
44 pawnAttacksTable[BLACK][square] = calculateBlackPawnAttacks(bb);
45 knightAttacksTable[square] = calculateKnightAttacks(bb);
46 kingAttacksTable[square] = calculateKingAttacks(bb);
47 }

50void initializeBetweenLookupTable() {
51 for (int from = 0; from < 64; from++) {
52 for (int to = 0; to < 64; to++) {
53 uint64_t m1 = -1ULL;
54 uint64_t a2a7 = 0x0001010101010100ULL;
55 uint64_t b2g7 = 0x0040201008040200ULL;
56 uint64_t h1b7 = 0x0002040810204080ULL;
57 uint64_t btwn, line, rank, file;
58
59 btwn = m1 << from ^ m1 << to;
60 file = (to & 7) - (from & 7);
61 rank = ((to | 7) - from) >> 3;
62 line = (file & 7) - 1 & a2a7;
63 line += 2 * (((rank & 7) - 1) >> 58);
64 line += (rank - file & 15) - 1 & b2g7;
65 line += (rank + file & 15) - 1 & h1b7;
66 line *= btwn & -btwn;
67
68 betweenLookupTable[from][to] = line & btwn;
69 }
70 }
71}

73void initializeLineLookupTable() {
74 for (uint8_t fromSquare = 0; fromSquare < SQUARES; ++fromSquare) {
75 for (uint8_t toSquare = 0; toSquare < SQUARES; ++toSquare) {
76 uint64_t line = 0;
77 uint64_t fromBb = squareToBitboard(fromSquare);
78 uint64_t toBb = squareToBitboard(toSquare);
79
80 if ((getRookAttacks(fromSquare, 0) & toBb) != 0) {
81 line = (getRookAttacks(fromSquare, 0) & getRookAttacks(toSquare, 0)) | fromBb | toBb;
82 } else if ((getBishopAttacks(fromSquare, 0) & toBb) != 0) {
83 line = (getBishopAttacks(fromSquare, 0) & getBishopAttacks(toSquare, 0)) | fromBb | toBb;
84 }
85
86 lineLookupTable[fromSquare][toSquare] = line;
87 }
88 }

133uint64_t getBishopAttacks(const uint8_t square, uint64_t occupied) {
134 assert(square < SQUARES);
135 occupied &= getBishopMask(square);
136 occupied *= getBishopMagic(square);
137 occupied >>= 64 - BBits[square];
138
139 return getBishopMagicAttacks(square, occupied);
140}

148uint64_t getRookAttacks(const uint8_t square, uint64_t occupied) {
149 assert(square < SQUARES);
150 occupied &= getRookMask(square);
151 occupied *= getRookMagic(square);
152 occupied >>= 64 - RBits[square];
153
154 return getRookMagicAttacks(square, occupied);
155}

163uint64_t queenAttacks(const uint8_t square, const uint64_t occupied) {
164 assert(square < SQUARES);
165 return getBishopAttacks(square, occupied) | getRookAttacks(square, occupied);
166}
