38void initializeAttackLookupTables() {
39 for (uint8_t square = 0; square < SQUARES; ++square) {
40 const uint64_t bb = squareToBitboard(square);
41
42 pawnAttacksTable[WHITE][square] = calculateWhitePawnAttacks(bb);
43 pawnAttacksTable[BLACK][square] = calculateBlackPawnAttacks(bb);
44 knightAttacksTable[square] = calculateKnightAttacks(bb);
45 kingAttacksTable[square] = calculateKingAttacks(bb);
46 }
47}

49void initializeBetweenLookupTable() {
50 for (int from = 0; from < 64; from++) {
51 for (int to = 0; to < 64; to++) {
52 uint64_t m1 = -1ULL;
53 uint64_t a2a7 = 0x0001010101010100ULL;
54 uint64_t b2g7 = 0x0040201008040200ULL;
55 uint64_t h1b7 = 0x0002040810204080ULL;
56 uint64_t btwn, line, rank, file;
57
58 btwn = m1 << from ^ m1 << to;
59 file = (to & 7) - (from & 7);
60 rank = ((to | 7) - from) >> 3;
61 line = (file & 7) - 1 & a2a7;
62 line += 2 * (((rank & 7) - 1) >> 58);
63 line += (rank - file & 15) - 1 & b2g7;
64 line += (rank + file & 15) - 1 & h1b7;
65 line *= btwn & -btwn;
66
67 betweenLookupTable[from][to] = line & btwn;
68 }
69 }
70}

114uint64_t getBishopAttacks(const uint8_t square, uint64_t occupied) {
115 assert(square < SQUARES);
116 occupied &= getBishopMask(square);
117 occupied *= getBishopMagic(square);
118 occupied >>= 64 - BBits[square];
119
120 return getBishopMagicAttacks(square, occupied);
121}

129uint64_t getRookAttacks(const uint8_t square, uint64_t occupied) {
130 assert(square < SQUARES);
131 occupied &= getRookMask(square);
132 occupied *= getRookMagic(square);
133 occupied >>= 64 - RBits[square];

144uint64_t queenAttacks(const uint8_t square, const uint64_t occupied) {
145 assert(square < SQUARES);
146 return getBishopAttacks(square, occupied) | getRookAttacks(square, occupied);
